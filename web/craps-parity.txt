# Crapless craps settlement cases, shared by both builds.
#
# The Compose engine (CrapsRules.resolve) and the web engine (window.CR.resolve)
# are two hand-written copies of the same maths. This file is the referee:
# CrapsParityTest.kt runs it against Kotlin, web/cr-parity.html runs it against
# the JavaScript, and both must agree with every line here.
#
# point | bets | roll | newPoint returns | surviving bets
#
# "point" is the number the shooter is chasing, or "-" on the come out.
# Bets are a comma list, "-" for a bare felt:
#   pass=10   the line          odds=30  free odds behind it
#   come=10   waiting on the come        cN=flat:odds  a come bet on N
#   pN=12     placed on N        field=5
#   hN=5      the hard way of N
#   any7 craps aces ace2 yo box   the one-roll centre bets
#
# "returns" is everything coming back to the purse on this throw. Bets that win
# a decision — the line and the come — are paid and taken down, stake and all.
# Bets that ride are paid their winnings and left standing, so they show up
# again under "surviving".

# ---- the come out: seven wins it, and nothing loses it ----
- | pass=10 | 3+4 | - 20 | -
- | pass=10 | 1+1 | 2 0 | pass=10                    # snake eyes is a point here, not a loser
- | pass=10 | 1+2 | 3 0 | pass=10
- | pass=10 | 6+6 | 12 0 | pass=10
- | pass=10 | 5+6 | 11 0 | pass=10                   # nor is the yo a winner
- | pass=10 | 3+3 | 6 0 | pass=10

# ---- the point, and the free odds behind it ----
2 | pass=10,odds=30 | 1+1 | - 230 | -                # six to one on the two
11 | pass=10,odds=30 | 5+6 | - 140 | -               # three to one on the eleven
6 | pass=10,odds=30 | 3+3 | - 86 | -                 # six to five on the six
10 | pass=10,odds=30 | 5+5 | - 110 | -               # two to one on the ten

# ---- the seven that sweeps ----
6 | pass=10,odds=20,p6=12,p8=12,c5=10:20,h8=5 | 3+4 | - 0 | -
6 | pass=10,come=10 | 1+6 | - 20 | -                 # the waiting come bet still wins

# ---- place bets, asleep through a come out ----
- | pass=10,p6=12 | 3+3 | 6 0 | pass=10,p6=12
- | pass=10,p6=12 | 3+4 | - 20 | p6=12               # the come out seven cannot touch them
8 | p6=12 | 4+2 | 8 14 | p6=12                       # seven to six, and the chips stay
6 | p3=12 | 1+2 | 6 33 | p3=12                       # eleven to four on the three
6 | p12=8 | 6+6 | 6 44 | p12=8                       # eleven to two on the twelve

# ---- the field ----
6 | field=10 | 1+1 | 6 20 | field=10                 # the two pays double
6 | field=10 | 6+6 | 6 30 | field=10                 # the twelve pays triple
6 | field=10 | 1+3 | 6 10 | field=10
6 | field=10 | 2+3 | 6 0 | -                         # five is not in the field

# ---- the hardways ----
6 | h8=5 | 4+4 | 6 45 | h8=5
6 | h8=5 | 5+3 | 6 0 | -                             # the easy way kills it
6 | h4=5 | 2+2 | 6 35 | h4=5
6 | h8=5 | 3+4 | - 0 | -                             # and so does any seven

# ---- the come, travelling and paid ----
6 | come=10 | 4+5 | 6 0 | c9=10:0
6 | c9=10:20 | 4+5 | 6 70 | -                        # three to two behind the nine
6 | c9=10:0,come=10 | 4+5 | 6 20 | c9=10:0           # one pays, the next travels onto it

# ---- the centre, one throw at a time ----
6 | yo=5 | 5+6 | 6 75 | yo=5
6 | yo=5 | 5+5 | 6 0 | -
6 | craps=5,field=10 | 6+6 | 6 65 | field=10,craps=5
6 | any7=5 | 3+4 | - 20 | any7=5                     # it wins on the very seven that sevens out
6 | aces=5,box=5 | 1+1 | 6 150 | aces=5
